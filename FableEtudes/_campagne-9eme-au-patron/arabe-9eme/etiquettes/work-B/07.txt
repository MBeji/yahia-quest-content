=== PIEGES ===
PIEGE: الحكمُ على الكلمةِ من مكانِها لا من معناها. قال تلميذٌ عن «سافرَ الشّاعرُ إلى القيروانِ» في بيتٍ شعريٍّ: «مجازٌ، لأنّه شعرٌ». ويُغري بذلك أنّ الشّعرَ أكثرُ الكلامِ صورًا. والصّوابُ: الشّاعرُ سافرَ فعلًا، فالفعلُ على معناه الأصليِّ ← **حقيقةٌ**؛ فالحكمُ يُبنى على معنى الكلمةِ في جملتِها لا على نوعِ النّصِّ.
PIEGE: قلبُ الطّرفَين. قال تلميذٌ في «الجنديُّ كالصّخرةِ ثباتًا»: «المشبَّهُ هو الصّخرةُ، لأنّها أشهرُ في الثّباتِ». ويُغري بذلك أنّ المشبَّهَ به يكونُ في العادةِ أوضحَ في الصّفةِ من المشبَّهِ. والصّوابُ: المشبَّهُ هو **الجنديُّ**، لأنّه الموصوفُ الذي يدورُ عليه الكلامُ؛ و«الصّخرةُ» ذُكرتْ لتُوضّحَ ثباتَه.
PIEGE: الحكمُ بالاستعارةِ لأنّ الأداةَ غائبةٌ. قال تلميذٌ عن «العلمُ نورٌ»: «استعارةٌ، لأنّه لا كافَ فيها». ويُغري بذلك أنّ الاستعارةَ تُحذفُ فيها الأداةُ فعلًا. والصّوابُ: تُحذفُ فيها الأداةُ **وأحدُ الطّرفَين**؛ وفي «العلمُ نورٌ» الطّرفانِ مذكوران (العلمُ والنّورُ)، فهي تشبيهٌ. فعُدَّ الطّرفَين قبلَ أن تحكمَ.
PIEGE: شرحُ المعنى الحرفيِّ بدلَ تحليلِ الصّورةِ. قال تلميذٌ عن «ولا بدَّ لليلِ أن ينجلي»: «الليلُ وقتٌ بين الغروبِ والفجرِ، وانجلاؤه ذهابُ ظلامِه». ويُغري بذلك أنّ جوابَه صحيحٌ في ذاتِه. والصّوابُ: هذا وصفٌ للمعنى الأصليِّ لا للصّورةِ؛ فالجوابُ الكاملُ يقولُ ما أُريدَ بالليلِ (الظّلمُ)، وما جمعَ الطّرفَين، وما فعلتْه الصّورةُ في القارئِ.
PIEGE: عدُّ الصّورةِ حجّةً تُثبتُ الأطروحةَ. قال تلميذٌ: «قولُنا «لصٌّ صامتٌ» دليلٌ على أنّ التّدخينَ مضرٌّ». ويُغري بذلك أنّ الصّورةَ تؤثّرُ في السّامعِ فعلًا. والصّوابُ: الصّورةُ تُحرّكُ الوجدانَ وتقرّبُ الفكرةَ المجرّدةَ من المحسوسِ كما يفعلُ المثالُ، لكنّها لا تُثبتُ؛ فالإثباتُ للحجّةِ (علميّةٍ أو واقعيّةٍ أو منطقيّةٍ…)، والصّورةُ تُعينُها ولا تقومُ مقامَها.
=== quiz (اختبار فهم الدرس) ===
-- quiz#1 [mcq] d1 ما أركانُ التّشبيهِ الأربعةُ؟
   a) المشبَّهُ، والمشبَّهُ به، وأداةُ التّشبيهِ، وأثرُ الصّورةِ
   b) المشبَّهُ، والمشبَّهُ به، ووجهُ الشّبَهِ، والمعنى المجازيُّ
  *c) المشبَّهُ، والمشبَّهُ به، وأداةُ التّشبيهِ، ووجهُ الشّبَهِ
   d) المشبَّهُ، وأداةُ التّشبيهِ، ووجهُ الشّبَهِ، والمعنى الحقيقيُّ
  EXPL: أركانُ التّشبيهِ أربعةٌ: المشبَّهُ (ما نصفُه)، والمشبَّهُ به (ما نقارنُه به)، وأداةُ التّشبيهِ (الكلمةُ الرّابطةُ)، ووجهُ الشّبَهِ (الصّفةُ المشتركةُ). أمّا أثرُ الصّورةِ والمعنى المجازيُّ والمعنى الحقيقيُّ فمن مفرداتِ الدّرسِ، لكنّها ليستْ من أركانِ التّشبيهِ.
-- quiz#2 [mcq] d1 في أيِّ الجملِ التّاليةِ استُعملَ الفعلُ «نامَ» في غيرِ معناهُ الحقيقيِّ؟
  *a) «نامتِ المدينةُ بعدَ منتصفِ اللّيلِ»
   b) «نامَ الرّاعي تحتَ الشّجرةِ الكبيرةِ ظهرًا»
   c) «نامَ المسافرُ في القطارِ طوالَ الرّحلةِ»
   d) «نامَ الطّفلُ في حِضنِ أمِّه بعدَ الرّضاعةِ»
  EXPL: «نامتِ المدينةُ» مجازٌ: النّومُ في معناه الأصليِّ حالُ كائنٍ حيٍّ يُغمضُ عينَيه، والمدينةُ بناءٌ لا عينَين له؛ فالمرادُ سكونُها وهدوءُ أهلِها، وبين المعنيَين شبهٌ. أمّا الرّاعي والمسافرُ والطّفلُ فكلٌّ منهم يغمضُ عينَيه فعلًا، فالفعلُ في جملِهم على حقيقتِه.
-- quiz#3 [mcq] d2 ما نوعُ الصّورةِ في قولِنا: «الصّبرُ مفتاحُ الفرجِ»؟
   a) استعارةٌ، لأنّ الأداةَ غائبةٌ فيها وفي وجهِ الشّبَهِ
   b) استعارةٌ، لأنّ المشبَّهَ به محذوفٌ ولم يبقَ إلّا المشبَّهُ
   c) حقيقةٌ، لأنّ الصّبرَ يفتحُ الأبوابَ المغلقةَ فعلًا
  *d) تشبيهٌ، لأنّ الطّرفَين مذكوران وإن غابتِ الأداةُ
  EXPL: ذُكرَ الطّرفانِ: «الصّبرُ» (المشبَّهُ) و«مفتاحُ الفرجِ» (المشبَّهُ به)، ولم تُذكرِ الأداةُ ولا وجهُ الشّبَهِ؛ والتّشبيهُ يبقى تشبيهًا ما دامَ طرفاهُ معًا. ولو حُذفَ أحدُ الطّرفَين لصارتِ العبارةُ استعارةً، وليس المحذوفُ هنا المشبَّهَ به، فهو «مفتاحُ الفرجِ» المذكورُ.
-- quiz#4 [mcq] d2 ما الاستعارةُ كما يُعرّفُها كتابُ النّصوصِ؟
   a) تشبيهٌ حُذفتْ أداتُه ووجهُ شبهِه معًا وبقيَ طرفاهُ كاملَين في الكلامِ
  *b) تشبيهٌ حُذفتْ كلُّ أركانِه ما عدا المشبَّهَ أو المشبَّهَ به
   c) كلمةٌ تُستعملُ في معناها الأصليِّ بعدَ حذفِ المشبَّهِ به
   d) مقارنةٌ بين شيئَين تُذكرُ فيها الأركانُ الأربعةُ كلُّها
  EXPL: الاستعارةُ تشبيهٌ لم يبقَ من أركانِه إلّا طرفٌ واحدٌ: المشبَّهُ أو المشبَّهُ به. فإن بقيَ الطّرفانِ معًا — ولو حُذفتِ الأداةُ ووجهُ الشّبَهِ — فهو تشبيهٌ لا استعارةٌ، وإن ذُكرتِ الأركانُ كلُّها فهو تشبيهٌ تامُّ الأركانِ.
-- quiz#5 [mcq] d2 ما الذي أضافتْه الصّورةُ في «ولا بدَّ لليلِ أن ينجلي» إلى القولِ الصّريحِ «لا بدَّ للظّلمِ أن ينتهيَ»؟
   a) ذكرتْ دليلًا علميًّا يُثبتُ بالأرقامِ أنّ الظّلمَ سينتهي قريبًا
   b) حوّلتِ الكلامَ إلى وصفٍ لليلِ الطّبيعةِ وظلامِه وحدَه
  *c) جعلتِ الظّلمَ المجرّدَ مشهدًا محسوسًا، وبعثتِ الأملَ بفجرٍ آتٍ
   d) حذفتِ المشبَّهَ به وأبقتْ أداةَ التّشبيهِ في الكلامِ
  EXPL: الصّورةُ هنا استعارةٌ: الظّلمُ (المحذوفُ) صارَ ليلًا يُرى ويُحَسُّ، ولأنّ الليلَ مهما طالَ ينجلي بعثتِ الصّورةُ الأملَ في السّامعِ. وليستْ دليلًا علميًّا، فالصّورةُ تؤثّرُ ولا تُثبتُ؛ وليس الكلامُ عن ليلِ الطّبيعةِ، وقد بقيَ في العبارةِ المشبَّهُ به «اللّيلُ» لا الأداةُ.
=== 01 (تمرين: التعرّف على الصور البيانية) ===
-- 01#1 [mcq] d1 ما المشبَّهُ به في قولِنا: «خالدٌ كالبحرِ كرمًا»؟
   a) خالدٌ
  *b) البحرُ
   c) الكافُ
   d) كرمًا
  EXPL: «خالدٌ» هو المشبَّهُ، أي الموصوفُ؛ و«البحرُ» هو المشبَّهُ به، أي ما قورنَ به؛ والكافُ أداةُ التّشبيهِ؛ و«كرمًا» هو وجهُ الشّبَهِ، أي الصّفةُ المشتركةُ بين خالدٍ والبحرِ.
-- 01#2 [mcq] d1 ما أداةُ التّشبيهِ في قولِنا: «كأنَّ النّجومَ في السّماءِ دُرَرٌ منثورةٌ»؟
   a) في
   b) دُرَرٌ
   c) منثورةٌ
  *d) كأنَّ
  EXPL: «كأنَّ» هي الكلمةُ الرّابطةُ بين المشبَّهِ «النّجومُ» والمشبَّهِ به «دُرَرٌ منثورةٌ»، فهي أداةُ التّشبيهِ. أمّا «دُرَرٌ» فجزءٌ من المشبَّهِ به، و«في» حرفُ جرٍّ، و«منثورةٌ» وصفٌ للدُّررِ.
-- 01#3 [mcq] d1 في أيِّ الجملِ التّاليةِ استُعملَ الفعلُ «ضحكَ» في معناهُ الحقيقيِّ؟
  *a) «ضحكَ الطّفلُ لمّا رأى المهرِّجَ يتعثّرُ»
   b) «ضحكتِ الأرضُ بالأزهارِ في الرّبيعِ»
   c) «ضحكَ الصّباحُ فوقَ الحقولِ المخضرّةِ الواسعةِ»
   d) «ضحكتِ الشّمسُ للمدينةِ بعدَ المطرِ»
  EXPL: الضّحكُ في أصلِه فعلُ إنسانٍ يفتحُ فمَه فرحًا، و«ضحكَ الطّفلُ» على هذا المعنى فهو حقيقةٌ. أمّا الأرضُ والصّباحُ والشّمسُ فلا أفواهَ لها تضحكُ بها، فاستُعملَ الفعلُ فيها في معنًى غيرِ حقيقيٍّ هو الإزهارُ والإشراقُ والصّفاءُ.
-- 01#4 [mcq] d2 ما أداةُ التّشبيهِ وما وجهُ الشّبَهِ في قولِنا: «هو يُشبهُ الجبلَ رسوخًا»؟
   a) الأداةُ «يُشبهُ»، ووجهُ الشّبَهِ «الجبلَ»
   b) الأداةُ «هو»، ووجهُ الشّبَهِ «رسوخًا»
  *c) الأداةُ «يُشبهُ»، ووجهُ الشّبَهِ «رسوخًا»
   d) الأداةُ «الجبلَ»، ووجهُ الشّبَهِ «يُشبهُ»
  EXPL: الأداةُ قد تكونُ فعلًا مثلَ «يُشبهُ»، وهي هنا الكلمةُ الرّابطةُ بين المشبَّهِ «هو» والمشبَّهِ به «الجبلَ». ووجهُ الشّبَهِ هو الصّفةُ المشتركةُ «رسوخًا». أمّا «الجبلَ» فمشبَّهٌ به لا وجهُ شبهٍ، و«هو» مشبَّهٌ لا أداةٌ.
-- 01#5 [mcq] d2 أيُّ العباراتِ التّاليةِ تشبيهٌ لا استعارةٌ؟
   a) «رأيتُ بحرًا يمشي على اليابسةِ»
  *b) «المدرسةُ بيتُنا الثّاني»
   c) «بكتِ الغيومُ فوقَ الجبالِ»
   d) «أطلَّ بدرٌ من نافذةِ القسمِ»
  EXPL: في «المدرسةُ بيتُنا الثّاني» الطّرفانِ مذكوران: «المدرسةُ» (المشبَّهُ) و«بيتُنا» (المشبَّهُ به)، فهي تشبيهٌ غابتْ أداتُه. وفي العباراتِ الأخرى لم يبقَ إلّا طرفٌ واحدٌ: «بحرًا» أو «الغيومُ» أو «بدرٌ»، وهذا شرطُ الاستعارةِ.
-- 01#6 [mcq] d2 في قولِنا «سمعتُ بلبلًا يُنشدُ في حفلِ المدرسةِ» (والمقصودُ تلميذٌ جميلُ الصّوتِ) استعارةٌ. أيُّ ركنٍ من أركانِ التّشبيهِ بقيَ فيها؟
   a) المشبَّهُ، وهو التّلميذُ المنشدُ
   b) أداةُ التّشبيهِ، وهي الكافُ
   c) وجهُ الشّبَهِ، وهو جمالُ الصّوتِ
  *d) المشبَّهُ به، وهو البلبلُ
  EXPL: المذكورُ في العبارةِ «بلبلًا»، وهو المشبَّهُ به. أمّا التّلميذُ (المشبَّهُ) والأداةُ ووجهُ الشّبَهِ (جمالُ الصّوتِ) فقد حُذفتْ، وأصلُ العبارةِ «سمعتُ تلميذًا كالبلبلِ صوتًا يُنشدُ في حفلِ المدرسةِ».
=== 02 (⚔️ زعيم الفصل: تحليل الصور البيانية المعمَّقة) ===
-- 02#1 [mcq] d2 في قولِنا «أطلَّ بدرٌ من نافذةِ القسمِ» يُقصدُ تلميذٌ جميلُ الوجهِ. ما التّشبيهُ الذي اختُصرتْ منه هذه الاستعارةُ؟
  *a) «أطلَّ تلميذٌ كالبدرِ حُسنًا من نافذةِ القسمِ»
   b) «أطلَّ بدرٌ كالتّلميذِ حُسنًا من نافذةِ القسمِ»
   c) «أطلَّ بدرٌ حقيقيٌّ من نافذةِ القسمِ»
   d) «أطلَّ تلميذٌ حَسنُ الوجهِ من نافذةِ القسمِ»
  EXPL: المشبَّهُ المحذوفُ «التّلميذُ»، والمشبَّهُ به الباقي «البدرُ»، والأداةُ الكافُ، ووجهُ الشّبَهِ «الحُسنُ»؛ فالأصلُ «أطلَّ تلميذٌ كالبدرِ حُسنًا». وقلبُ الطّرفَين يجعلُ البدرَ هو الموصوفَ، والبدرُ الحقيقيُّ لا يُطلُّ من نافذةِ قسمٍ، وصفةُ «حَسنُ الوجهِ» وحدَها لا مقارنةَ فيها.
-- 02#2 [mcq] d2 في قولِنا «بكتِ الغيومُ فوقَ الجبالِ» استعارةٌ. أيُّ ركنٍ من أركانِ التّشبيهِ بقيَ فيها؟
   a) المشبَّهُ به، وهو «الغيومُ»
   b) وجهُ الشّبَهِ، وهو الفعلُ «بكتْ»
  *c) المشبَّهُ، وهو «الغيومُ»
   d) المشبَّهُ به، وهو «الجبالُ»
  EXPL: الغيومُ هي الموصوفُ، أي المشبَّهُ، وقد بقيتْ في الكلامِ؛ أمّا المشبَّهُ به — الإنسانُ الباكي — فحُذفَ، وفعلُ «بكتْ» يدلُّ عليه وليس هو وجهَ الشّبَهِ. و«الجبالُ» مكانُ البكاءِ لا طرفٌ في المقارنةِ. وهذه الحالةُ يشملُها تعريفُ الكتابِ: «حُذفتْ كلُّ أركانِه ما عدا المشبَّهَ أو المشبَّهَ به».
-- 02#3 [mcq] d3 أيُّ التّحليلاتِ التّاليةِ يعيّنُ أركانَ التّشبيهِ تعيينًا صحيحًا في المثلِ: «الوقتُ كالسّيفِ إنْ لم تقطعْهُ قطعَكَ»؟
   a) المشبَّهُ «الوقتُ»، والمشبَّهُ به «السّيفُ»، والأداةُ «كالسّيفِ»، ووجهُ الشّبَهِ «القطعُ»
   b) المشبَّهُ «السّيفُ»، والمشبَّهُ به «الوقتُ»، والأداةُ الكافُ، ووجهُ الشّبَهِ «القطعُ»
   c) المشبَّهُ «الوقتُ»، والمشبَّهُ به «قطعَكَ»، والأداةُ الكافُ، ووجهُ الشّبَهِ «السّيفُ»
  *d) المشبَّهُ «الوقتُ»، والمشبَّهُ به «السّيفُ»، والأداةُ الكافُ، ووجهُ الشّبَهِ «القطعُ»
  EXPL: الموصوفُ هو «الوقتُ» فهو المشبَّهُ، وما قورنَ به «السّيفُ» فهو المشبَّهُ به، والكافُ حرفٌ مستقلٌّ في التّحليلِ هو الأداةُ ولا يدخلُ في المشبَّهِ به؛ ووجهُ الشّبَهِ «القطعُ» الذي دلّتْ عليه جملةُ «إنْ لم تقطعْهُ قطعَكَ». وقلبُ الطّرفَين، أو إدخالُ الكافِ في المشبَّهِ به، أو جعلُ «قطعَكَ» مشبَّهًا به، أخطاءٌ في التّعيينِ.
-- 02#4 [mcq] d3 يقولُ المعلّمُ عن عبارتَين: «الجنديُّ أسدٌ في المعركةِ» و«قاتلَ الأسدُ دفاعًا عن الحدودِ» (والمقصودُ جنديٌّ). أيُّ الأحكامِ التّاليةِ صحيحٌ؟
   a) كلتاهما استعارةٌ، لأنّ الأداةَ غائبةٌ فيهما معًا
  *b) «الجنديُّ أسدٌ…» تشبيهٌ لذكرِ طرفَيه، و«قاتلَ الأسدُ…» استعارةٌ لحذفِ المشبَّهِ
   c) «الجنديُّ أسدٌ…» استعارةٌ لغيابِ الأداةِ، و«قاتلَ الأسدُ…» تشبيهٌ لذكرِ المشبَّهِ به
   d) كلتاهما تشبيهٌ، لأنّ كلتَيهما تذكرُ الأسدَ في الكلامِ
  EXPL: في «الجنديُّ أسدٌ في المعركةِ» ذُكرَ الطّرفانِ (الجنديُّ والأسدُ) فهي تشبيهٌ غابتْ أداتُه. وفي «قاتلَ الأسدُ دفاعًا عن الحدودِ» لم يُذكرِ الجنديُّ (المشبَّهُ) وبقيَ الأسدُ (المشبَّهُ به) وحدَه، فهي استعارةٌ. وغيابُ الأداةِ وحدَه لا يصنعُ استعارةً، وذكرُ المشبَّهِ به وحدَه ليس تشبيهًا.
-- 02#5 [mcq] d3 قالَ كاتبٌ في مقالٍ عن الإدمانِ: «الإدمانُ سلسلةٌ تلتفُّ حولَ معصمِ الشّابِّ». ما غايةُ الكاتبِ من هذه العبارةِ؟
   a) الإقناعُ بحجّةٍ علميّةٍ تُثبتُ أضرارَ الإدمانِ على الجسمِ كلِّه
   b) الإقناعُ بحجّةٍ رقميّةٍ تُحصي عددَ المدمنين في البلادِ
  *c) التّأثيرُ في وجدانِ القارئِ بصورةٍ تُشعرُه بفقدانِ الحرّيّةِ
   d) الإقناعُ بحجّةٍ تاريخيّةٍ تروي نشأةَ الإدمانِ عبرَ العصورِ
  EXPL: العبارةُ تشبيهٌ: المشبَّهُ «الإدمانُ» والمشبَّهُ به «سلسلةٌ»، ولا تقدّمُ خبرًا علميًّا ولا رقمًا ولا واقعةً تاريخيّةً؛ فهي لا تخاطبُ العقلَ بدليلٍ، وإنّما تخاطبُ وجدانَ القارئِ فتُشعرُه بالقيدِ، وهذه وظيفةُ التّأثيرِ.
-- 02#6 [mcq] d3 قالَ تلميذٌ في «أمّي كالشّمعةِ تحترقُ لتُضيءَ لنا»: «المشبَّهُ هو الشّمعةُ، لأنّها أشهرُ في الإضاءةِ». ما الخطأُ في كلامِه؟
  *a) المشبَّهُ هو الموصوفُ «أمّي»، أمّا الشّمعةُ فهي المشبَّهُ به
   b) العبارةُ استعارةٌ لا تشبيهٌ، فلا مشبَّهَ فيها أصلًا ولا مشبَّهَ به
   c) المشبَّهُ هو «تحترقُ»، لأنّها الصّفةُ المشتركةُ بين الطّرفَين
   d) لا خطأَ، فالمشبَّهُ هو الأشهرُ دائمًا في الصّفةِ المشتركةِ بينهما
  EXPL: المشبَّهُ ما نتحدّثُ عنه ونصفُه، وهو «أمّي»؛ والمشبَّهُ به ما نستعينُ به لإيضاحِ الوصفِ، وهو «الشّمعةُ»، ولو كانتْ هي الأشهرَ في الإضاءةِ. والعبارةُ تشبيهٌ لأنّ الطّرفَين مذكوران، و«تحترقُ» جزءٌ من وجهِ الشّبَهِ لا مشبَّهٌ. فقاعدةُ «الأشهرُ هو المشبَّهُ» عكسُ الصّوابِ.
=== 04 (👑 تحدّي النخبة: الصور البيانية في نصوص أدبية) ===
-- 04#1 [mcq] d2 قالَ امرؤُ القيسِ: «وَلَيْلٍ كَمَوْجِ البَحْرِ أَرْخَى سُدُولَهُ ✻ عَلَيَّ بِأَنْوَاعِ الهُمُومِ لِيَبْتَلِي». ما الصّفةُ التي تجمعُ بين الليلِ وموجِ البحرِ في هذا التّشبيهِ، ولم يذكرْها الشّاعرُ؟
   a) اللّمعانُ وصفاءُ اللّونِ
   b) السّرعةُ وخفّةُ الحركةِ
   c) الجمالُ وهدوءُ الحالِ
  *d) الهولُ وكثافةُ الظّلمةِ
  EXPL: المشبَّهُ «ليلٍ»، والمشبَّهُ به «موجِ البحرِ»، والأداةُ الكافُ، ووجهُ الشّبَهِ غيرُ مذكورٍ فنستنتجُه من السّياقِ: الموجُ يتتابعُ مرعبًا فيحجبُ ما وراءَه، والليلُ يُرخي ظلامَه ثقيلًا «بأنواعِ الهمومِ»؛ فالجامعُ الهولُ وكثافةُ الظّلمةِ. أمّا اللّمعانُ والصّفاءُ والخفّةُ والجمالُ والهدوءُ فتناقضُ ما يصفُه الشّاعرُ من همٍّ وثقلٍ.
-- 04#2 [mcq] d3 قالَ رسولُ اللهِ صلّى اللهُ عليه وسلّمَ: «المؤمنُ للمؤمنِ كالبنيانِ يشدُّ بعضُه بعضًا». أيُّ التّحليلاتِ التّاليةِ يعيّنُ أركانَ التّشبيهِ تعيينًا صحيحًا؟
   a) المشبَّهُ «البنيانُ»، والمشبَّهُ به «المؤمنُ»، والأداةُ الكافُ، ووجهُ الشّبَهِ «يشدُّ بعضُه بعضًا»
  *b) المشبَّهُ «المؤمنُ»، والمشبَّهُ به «البنيانُ»، والأداةُ الكافُ، ووجهُ الشّبَهِ «يشدُّ بعضُه بعضًا»
   c) المشبَّهُ «المؤمنُ»، والمشبَّهُ به «بعضُه»، والأداةُ الكافُ، ووجهُ الشّبَهِ «البنيانُ»
   d) المشبَّهُ «المؤمنُ»، والمشبَّهُ به «البنيانُ»، ولا أداةَ فيه، ووجهُ الشّبَهِ «للمؤمنِ»
  EXPL: الحديثُ يتحدّثُ عن المؤمنِ فهو المشبَّهُ، وقورنَ بالبنيانِ فهو المشبَّهُ به، والكافُ أداةُ التّشبيهِ، ووجهُ الشّبَهِ هو الصّفةُ المشتركةُ التي صرّحَ بها الحديثُ: «يشدُّ بعضُه بعضًا»، أي التّماسكُ والتّعاضدُ. وفي التّحليلاتِ الأخرى قلبٌ للطّرفَين، أو جعلُ «بعضُه» مشبَّهًا به، أو إنكارٌ للكافِ وهي مذكورةٌ في «كالبنيانِ».
-- 04#3 [mcq] d3 قالَ عنترةُ وهو في المعركةِ يتذكّرُ حبيبتَه: «فَوَدِدْتُ تَقْبِيلَ السُّيُوفِ لِأَنَّهَا ✻ لَمَعَتْ كَبَارِقِ ثَغْرِكِ المُتَبَسِّمِ». ما أثرُ هذه الصّورةِ في المعنى؟
  *a) تجعلُ السّيفَ محبوبًا، فيودُّ تقبيلَه لأنّ لمعانَه يذكّرُه بابتسامةِ حبيبتِه
   b) تُثبتُ بالبرهانِ أنّ السّيوفَ أشدُّ لمعانًا من الأسنانِ كلِّها في ساحةِ المعركةِ
   c) تُحوّلُ الكلامَ إلى وصفٍ علميٍّ لأثرِ الضّوءِ في المعادنِ اللّامعةِ
   d) تجعلُ الحبيبةَ محاربةً تحملُ سيفًا لامعًا في ساحةِ المعركةِ والدّماءِ
  EXPL: التّشبيهُ هنا: المشبَّهُ السّيوفُ، والمشبَّهُ به «بارقُ ثغرِكِ المتبسّمِ»، والأداةُ الكافُ، ووجهُ الشّبَهِ اللّمعانُ. وأثرُه أنّه جمعَ مشهدَ الحربِ بمشهدِ الحبِّ: فصارَ السّيفُ المخيفُ محبوبًا يودُّ الشّاعرُ تقبيلَه لأنّه يذكّرُه بابتسامةِ حبيبتِه. وهو صورةٌ تؤثّرُ في الوجدانِ ولا تُقدّمُ برهانًا ولا وصفًا علميًّا؛ والحبيبةُ ليستْ محاربةً، فالمشبَّهُ به «بارقُ ثغرِها» لا هي.
-- 04#4 [mcq] d3 قالَ البحتريُّ: «أَتَاكَ الرَّبِيعُ الطَّلْقُ يَخْتَالُ ضَاحِكًا ✻ مِنَ الحُسْنِ حَتَّى كَادَ أَنْ يَتَكَلَّمَا». أيُّ ركنٍ من أركانِ التّشبيهِ بقيَ في الشّطرِ الأوّلِ، وماذا أفادتِ الصّورةُ؟
   a) بقيَ المشبَّهُ به «الرّبيعُ»، وأفادتِ الصّورةُ أنّ الفصلَ حيٌّ فرِحٌ يعتزُّ بجمالِه كالإنسانِ
   b) بقيَ المشبَّهُ «الرّبيعُ»، وأفادتِ الصّورةُ دليلًا علميًّا على أنّ الرّبيعَ أجملُ فصولِ السّنةِ
  *c) بقيَ المشبَّهُ «الرّبيعُ»، وأفادتِ الصّورةُ أنّ الفصلَ حيٌّ فرِحٌ يعتزُّ بجمالِه كالإنسانِ
   d) بقيَ المشبَّهُ «الرّبيعُ»، ولم تُضفِ الصّورةُ شيئًا إلى القولِ الصّريحِ «جاءَ الرّبيعُ جميلًا»
  EXPL: الرّبيعُ هو الموصوفُ فهو المشبَّهُ، وقد بقيَ في الكلامِ؛ أمّا المشبَّهُ به — الإنسانُ المختالُ الضّاحكُ — فحُذفَ ودلَّ عليه «يختالُ ضاحكًا». وأثرُ الصّورةِ أنّها أحيتِ الفصلَ فجعلتْه كائنًا فرِحًا يعتزُّ بحُسنِه حتّى يكادَ ينطقُ، وهذا ما لا يقولُه «جاءَ الرّبيعُ جميلًا». وليس الرّبيعُ مشبَّهًا به هنا، والصّورةُ تؤثّرُ ولا تقدّمُ دليلًا علميًّا.
-- 04#5 [mcq] d3 قالَ الشّابّيُّ: «وَمَنْ لَا يُحِبُّ صُعُودَ الجِبَالِ ✻ يَعِشْ أَبَدَ الدَّهْرِ بَيْنَ الحُفَرْ». يريدُ أنّ مَن لا يسعى إلى المعالي يعشْ ذليلًا. ما نوعُ الصّورةِ في «صعودِ الجبالِ»؟
   a) حقيقةٌ، لأنّ الشّاعرَ يتحدّثُ عن متسلّقي الجبالِ ومشقّةِ الطّريقِ فيها
   b) تشبيهٌ، لأنّ الطّرفَين مذكوران في البيتِ معًا، وهما الجبالُ والحُفَرُ
   c) استعارةٌ، حُذفَ فيها المشبَّهُ به (السّعيُ إلى المعالي) وبقيَ المشبَّهُ (صعودُ الجبالِ)
  *d) استعارةٌ، حُذفَ فيها المشبَّهُ (السّعيُ إلى المعالي) وبقيَ المشبَّهُ به (صعودُ الجبالِ)
  EXPL: «صعودُ الجبالِ» لم يُستعملْ في معناه الحقيقيِّ، لأنّ الشّاعرَ لا يتحدّثُ عن رياضةٍ؛ وأصلُ الصّورةِ «السّعيُ إلى المعالي كصعودِ الجبالِ في المشقّةِ والارتفاعِ». حُذفَ المشبَّهُ (السّعيُ إلى المعالي) والأداةُ ووجهُ الشّبَهِ، وبقيَ المشبَّهُ به «صعودُ الجبالِ»، فهي استعارةٌ. وليستْ تشبيهًا، فـ«الجبالُ» و«الحُفَرُ» لا يُقارَنُ أحدُهما بالآخرِ بل يتقابلانِ: علوٌّ وانحدارٌ.
-- 04#6 [mcq] d3 قالَ أبو فراسٍ الحمدانيُّ: «سَيَذْكُرُنِي قَوْمِي إِذَا جَدَّ جِدُّهُمْ ✻ وَفِي اللَّيْلَةِ الظَّلْمَاءِ يُفْتَقَدُ البَدْرُ». ما المعنى المرادُ بـ«اللّيلةِ الظّلماءِ» و«البدرِ» في كلامِه؟
   a) «اللّيلةُ الظّلماءُ» وقتُ الرّاحةِ والأمانِ، و«البدرُ» الشّخصُ الذي يُهتدى به ويُحتاجُ إليه
  *b) «اللّيلةُ الظّلماءُ» وقتُ الشّدّةِ والخطرِ، و«البدرُ» الشّخصُ الذي يُهتدى به ويُحتاجُ إليه
   c) «اللّيلةُ الظّلماءُ» وقتُ الشّدّةِ والخطرِ، و«البدرُ» الشّخصُ الذي يُتخلّى عنه ولا يُحتاجُ إليه
   d) «اللّيلةُ الظّلماءُ» و«البدرُ» كلاهما على حقيقتِه، فالشّاعرُ يصفُ ليلةً مظلمةً غابَ فيها القمرُ
  EXPL: الشّاعرُ يقولُ إنّ قومَه سيذكرونَه عند الشّدائدِ، فاستعارَ «اللّيلةَ الظّلماءَ» للشّدّةِ، و«البدرَ» للرّجلِ الذي يُهتدى به؛ وحُذفَ المشبَّهانِ وبقيَ المشبَّهُ بهما، والمعنى أنّ الحاجةَ إليه في الشّدّةِ كحاجةِ السّائرِ في الظّلامِ إلى القمرِ. وهما مجازٌ لا حقيقةٌ، لأنّ الكلامَ عن مكانةِ رجلٍ عند قومِه لا عن طقسِ ليلةٍ.
=== 05 (📝 تدريب مراجعة: الصور البيانية) ===
-- 05#1 [mcq] d1 متى تكونُ الكلمةُ مجازًا؟
   a) إذا كانتْ غريبةً قديمةً لا يعرفُها أكثرُ القرّاءِ اليومَ ولا يستعملونَها
   b) إذا وردتْ في بيتِ شعرٍ أو في قصيدةٍ طويلةٍ كاملةٍ من القصائدِ
  *c) إذا استُعملتْ في معنًى غيرِ حقيقيٍّ بينَه وبين معناها الأصليِّ شبهٌ
   d) إذا حُذفتْ منها أداةُ التّشبيهِ ووجهُ الشّبَهِ معًا في الجملةِ
  EXPL: المجازُ استعمالُ الكلمةِ في معنًى غيرِ حقيقيٍّ يشبهُ معناها الأصليَّ: كـ«بكتِ السّماءُ». ولا تكونُ الكلمةُ مجازًا لأنّها غريبةٌ أو قديمةٌ، ولا لأنّها وردتْ في شعرٍ، فقد تردُ الكلمةُ على حقيقتِها في القصيدةِ؛ وحذفُ الأداةِ شأنُ التّشبيهِ والاستعارةِ لا تعريفُ المجازِ.
-- 05#2 [mcq] d2 أيُّ الأركانِ يجوزُ حذفُه من التّشبيهِ دونَ أن تتوقّفَ العبارةُ عن كونِها تشبيهًا؟
  *a) الأداةُ ووجهُ الشّبَهِ
   b) المشبَّهُ ووجهُ الشّبَهِ
   c) المشبَّهُ به والأداةُ
   d) المشبَّهُ والمشبَّهُ به
  EXPL: في التّشبيهِ طرفانِ لا بدَّ من ذكرِهما: المشبَّهُ والمشبَّهُ به. أمّا الأداةُ ووجهُ الشّبَهِ فيجوزُ حذفُهما أو حذفُ أحدِهما، كما في «العلمُ نورٌ». فإن حُذفَ المشبَّهُ أو المشبَّهُ به لم يبقَ تشبيهٌ بل استعارةٌ.
-- 05#3 [mcq] d2 ما الترتيبُ الصّحيحُ لخطواتِ قراءةِ الصّورةِ في بيتٍ؟
   a) بيِّنِ الأثرَ، ثمّ حدِّدِ الصّورةَ، ثمّ فكِّكْ أركانَها، ثمّ بيِّنِ المعنى الجامعَ
   b) حدِّدِ الصّورةَ، ثمّ بيِّنِ الأثرَ، ثمّ فكِّكْ أركانَها، ثمّ بيِّنِ المعنى الجامعَ
   c) فكِّكْ أركانَها، ثمّ بيِّنِ الأثرَ، ثمّ حدِّدِ الصّورةَ، ثمّ بيِّنِ المعنى الجامعَ
  *d) حدِّدِ الصّورةَ، ثمّ فكِّكْ أركانَها، ثمّ بيِّنِ المعنى الجامعَ، ثمّ بيِّنِ الأثرَ
  EXPL: نبدأ بتحديدِ الصّورةِ لأنّنا لا نفكّكُ ما لم نعرفْه؛ ثمّ نفكّكُ أركانَها لنعرفَ ما حُذفَ وما بقيَ؛ ثمّ نبيّنُ الصّفةَ الجامعةَ بين الطّرفَين؛ ونختمُ بالأثرِ لأنّه لا يُفهمُ قبلَ فهمِ الأركانِ والمعنى الجامعِ. وتقديمُ الأثرِ على الأركانِ يجعلُ الجوابَ انطباعًا بلا سندٍ.
-- 05#4 [mcq] d3 أيُّ العباراتِ التّاليةِ استعارةٌ؟
   a) «الدّنيا مدرسةٌ كبيرةٌ لأهلِها»
  *b) «غرّدَ الخطيبُ على المنبرِ»
   c) «هو كالجبلِ رسوخًا»
   d) «العلمُ كنزٌ لا يفنى»
  EXPL: في «غرّدَ الخطيبُ على المنبرِ» بقيَ طرفٌ واحدٌ هو المشبَّهُ «الخطيبُ»، وحُذفَ المشبَّهُ به (الطّائرُ) ودلَّ عليه فعلُ «غرّدَ»؛ فهي استعارةٌ. وفي «الدّنيا مدرسةٌ كبيرةٌ» و«العلمُ كنزٌ لا يفنى» الطّرفانِ مذكوران والأداةُ غائبةٌ، وفي «هو كالجبلِ رسوخًا» الأركانُ كلُّها مذكورةٌ؛ فكلُّها تشبيهٌ.
-- 05#5 [mcq] d3 في قولِ المتنبّي «فَقُلْتُ إِلَيْكَ إِنَّ مَعِي السَّحَابَا» يريدُ أميرًا كريمًا. ما الذي حُذفَ من التّشبيهِ الأصليِّ «إنّ معي أميرًا كالسّحابِ جُودًا»، وما الذي بقيَ؟
  *a) حُذفَ المشبَّهُ والأداةُ ووجهُ الشّبَهِ، وبقيَ المشبَّهُ به
   b) حُذفَ المشبَّهُ به والأداةُ ووجهُ الشّبَهِ، وبقيَ المشبَّهُ
   c) حُذفتِ الأداةُ ووجهُ الشّبَهِ فقط، وبقيَ الطّرفانِ معًا
   d) لم يُحذفْ شيءٌ، وبقيتِ الأركانُ الأربعةُ كلُّها
  EXPL: في الأصلِ المشبَّهُ «أميرًا» والأداةُ الكافُ والمشبَّهُ به «السّحابِ» ووجهُ الشّبَهِ «جُودًا». وفي عبارةِ الشّاعرِ لم يبقَ إلّا «السّحابا»، وهو المشبَّهُ به، فحُذفَ ما عداه، ولذلك هي استعارةٌ. ولو بقيَ الطّرفانِ لكانتْ تشبيهًا.
-- 05#6 [mcq] d3 ما موقعُ الصّورةِ من الحجاجِ؟
   a) تُثبتُ الأطروحةَ بالبرهانِ كما تفعلُ الحجّةُ العقليّةُ المنطقيّةُ
   b) تُحذفُ من النّصِّ الحجاجيِّ لأنّه يخاطبُ العقلَ وحدَه دونَ الوجدانِ
  *c) تؤثّرُ في الوجدانِ وتقرّبُ الفكرةَ، لكنّها لا تُثبتُ الأطروحةَ بذاتِها
   d) لا تدخلُ في الحجاجِ لأنّها خاصّةٌ بالشّعرِ وحدَه ولا تُستعملُ في النّثرِ
  EXPL: للحجاجِ وظيفتانِ: الإقناعُ بمخاطبةِ العقلِ والتّأثيرُ بمخاطبةِ الوجدانِ، والتّشبيهُ والمجازُ من أساليبِ البيانِ التي يستعينُ بها النّصُّ الحجاجيُّ؛ فالصّورةُ أقربُ إلى التّأثيرِ وتقرّبُ الفكرةَ المجرّدةَ، لكنّ الإثباتَ للحجّةِ. وليستِ الصّورةُ برهانًا، ولا تُحذفُ من الحجاجِ، ولا تختصُّ بالشّعرِ.
=== 06 (👑 تحدّي المناظرة ⭐⭐⭐⭐: الصّورة — تشبيه واستعارة وحقيقة ومجاز) ===
-- 06#1 [mcq] d2 قالَ تلميذٌ عن «سافرَ الشّاعرُ إلى القيروانِ» في بيتٍ شعريٍّ: «مجازٌ، لأنّه شعرٌ». ما الخطأُ في كلامِه؟
   a) الخطأُ أنّ الشّعرَ لا يكونُ فيه مجازٌ أبدًا، فهو كلامٌ يقولُ الحقائقَ كلَّها دونَ أيِّ صورةٍ
  *b) الخطأُ أنّ الحكمَ يُبنى على معنى الكلمةِ في جملتِها لا على نوعِ النّصِّ، والفعلُ هنا حقيقةٌ
   c) الخطأُ أنّ المجازَ هو الكلمةُ الغريبةُ، و«سافرَ» كلمةٌ مألوفةٌ يعرفُها كلُّ التّلاميذِ
   d) لا خطأَ، فكلُّ ما يردُ في القصيدةِ مجازٌ لأنّ الشّعرَ صورٌ كلُّه من أوّلِه إلى آخرِه
  EXPL: السّفرُ هنا فعلٌ حقيقيٌّ سافرَه الشّاعرُ، فالفعلُ مستعملٌ في معناه الأصليِّ وهو حقيقةٌ ولو وردَ في قصيدةٍ. فالحكمُ بالمجازِ يُبنى على معنى الكلمةِ في جملتِها لا على أنّها شعرٌ. وليس صحيحًا أنّ الشّعرَ خالٍ من المجازِ، ولا أنّ كلَّ ما في القصيدةِ مجازٌ، ولا طرفَين مقارنَين في الجملةِ.
-- 06#2 [mcq] d3 قالَ حافظٌ إبراهيمُ: «الأمُّ مدرسةٌ إذا أعددتَها ✻ أعددتَ شعبًا طيّبَ الأعراقِ». أيُّ التّحليلاتِ التّاليةِ للصّورةِ في الشّطرِ الأوّلِ صحيحٌ؟
   a) استعارةٌ لأنّ الأداةَ غائبةٌ، ووجهُ الشّبَهِ التّربيةُ والتّعليمُ؛ وأثرُها أنّها قرّبتْ دورَ الأمِّ من مؤسّسةٍ مألوفةٍ تُخرّجُ الأجيالَ
   b) تشبيهٌ مشبَّهُه «المدرسةُ» ومشبَّهُه به «الأمُّ»، ووجهُ الشّبَهِ التّربيةُ والتّعليمُ؛ وأثرُها أنّها قرّبتْ دورَ الأمِّ من مؤسّسةٍ مألوفةٍ تُخرّجُ الأجيالَ
   c) تشبيهٌ مشبَّهُه «الأمُّ» ومشبَّهُه به «مدرسةٌ»، ووجهُ الشّبَهِ التّربيةُ والتّعليمُ؛ وأثرُها أنّها ذكرتْ دليلًا علميًّا على أهمّيّةِ الأمِّ في المجتمعِ
  *d) تشبيهٌ مشبَّهُه «الأمُّ» ومشبَّهُه به «مدرسةٌ»، ووجهُ الشّبَهِ التّربيةُ والتّعليمُ؛ وأثرُها أنّها قرّبتْ دورَ الأمِّ من مؤسّسةٍ مألوفةٍ تُخرّجُ الأجيالَ
  EXPL: ذُكرَ الطّرفانِ (الأمُّ ومدرسةٌ) فهي تشبيهٌ ولو غابتِ الأداةُ، والموصوفُ هو الأمُّ فهي المشبَّهُ. والجامعُ بينهما أنّ كلًّا منهما تُنشئُ الأجيالَ وتُعلّمُها، وأثرُ الصّورةِ أنّها جعلتْ إعدادَ الأمِّ مؤسّسةً تصنعُ «شعبًا طيّبَ الأعراقِ». وفي التّحليلاتِ الأخرى خطأٌ في النّوعِ، أو قلبٌ للطّرفَين، أو جعلُ الصّورةِ دليلًا علميًّا وهي لا تُثبتُ.
-- 06#3 [mcq] d3 طُلبَ من تلميذٍ أن يحلّلَ «ولا بدَّ لليلِ أن ينجلي» ويبيّنَ أثرَ الصّورةِ، فكتبَ: «استعارةٌ، لأنّ الليلَ وقتٌ بين الغروبِ والفجرِ وانجلاؤه ذهابُ ظلامِه». ما الذي ينقصُ جوابَه؟
  *a) بيانُ ما أُريدَ بالليلِ (الظّلمُ) وما أضافتْه الصّورةُ في القارئِ
   b) أن يقولَ إنّها تشبيهٌ لا استعارةٌ لأنّ في العبارةِ طرفَين مذكورَين معًا
   c) أن يذكرَ دليلًا علميًّا على انجلاءِ الليلِ كلَّ صباحٍ
   d) لا ينقصُه شيءٌ، فقد شرحَ المعنى الأصليَّ شرحًا كاملًا
  EXPL: الجوابُ وصفَ المعنى الأصليَّ للّيلِ ولم يحلّلِ الصّورةَ؛ فلا هو بيّنَ أنّ الليلَ يُرادُ به الظّلمُ (وأصلُها «الظّلمُ كاللّيلِ»)، ولا بيّنَ الأثرَ: أنّها جعلتِ الظّلمَ مشهدًا يُرى وبعثتْ أملًا بفجرٍ آتٍ. وهي استعارةٌ لا تشبيهٌ لأنّ طرفًا واحدًا بقيَ، وليس المطلوبُ دليلًا علميًّا.
-- 06#4 [mcq] d3 سُئلَ تلميذٌ: «شبّهَ كاتبٌ القمارَ بالسّرابِ. هل أعجبَك التّشبيهُ والصّورةُ النّاشئةُ عنه؟ لِمَ؟». أيُّ التّعليلاتِ التّاليةِ أنسبُ؟
   a) أعجبَني لأنّه تشبيهٌ، والتّشبيهُ في كلِّ نصٍّ أجملُ من القولِ الصّريحِ دائمًا وأبدًا
   b) أعجبَني لأنّ كاتبَه مشهورٌ في بلدِنا، فلا بدَّ أن يكونَ كلامُه جميلًا
  *c) أعجبَني لأنّه جعلَ القمارَ يُغري من بعيدٍ ثمّ يخذلُ كما يخذلُ السّرابُ الظّمآنَ
   d) أعجبَني لأنّ كلمةَ «السّرابِ» غريبةٌ لا يعرفُها إلّا القليلُ
  EXPL: التّعليلُ الجيّدُ للإعجابِ بصورةٍ يُبنى على ما أضافتْه إلى المعنى: هنا قرّبتِ الصّورةُ خديعةَ القمارِ من مشهدٍ نعرفُه — ظمآنُ يجري وراءَ ماءٍ يلمعُ من بعيدٍ ثمّ لا يجدُ إلّا الرّملَ — فاتّضحَ المعنى وتأثّرَ القارئُ. أمّا مجرّدُ أنّه تشبيهٌ، أو أنّ كاتبَه مشهورٌ، أو أنّ كلمةً فيه غريبةٌ، فليستْ تعليلاتٍ تُبيّنُ ما أضافتْه الصّورةُ.
-- 06#5 [mcq] d3 قالَ تلميذٌ في فقرةٍ حجاجيّةٍ ضدَّ التّدخينِ: «أثبتُّ أطروحتي بقولي: التّدخينُ لصٌّ صامتٌ يسرقُ أنفاسَكَ». ما الخطأُ في كلامِه؟
   a) الخطأُ أنّه حذفَ أداةَ التّشبيهِ، ولا يكونُ التّشبيهُ حجّةً إلّا إذا ذُكرتْ أداتُه
   b) الخطأُ أنّ الصّورةَ ممنوعةٌ في الحجاجِ كلِّه، فلا تُستعملُ فيه إلّا الأرقامُ والوقائعُ
   c) لا خطأَ، فكلُّ صورةٍ تردُ في الحجاجِ تُعدُّ برهانًا على الأطروحةِ نفسِها
  *d) الخطأُ أنّ الصّورةَ تؤثّرُ في الوجدانِ ولا تُثبتُ؛ فالإثباتُ يحتاجُ إلى حجّةٍ
  EXPL: «لصٌّ صامتٌ» تشبيهٌ (المشبَّهُ التّدخينُ والمشبَّهُ به لصٌّ) يخاطبُ الوجدانَ فيُخيفُ السّامعَ؛ وهذا من وظيفةِ التّأثيرِ لا الإقناعِ. فهو يُعينُ الحجّةَ ولا يقومُ مقامَها، والإثباتُ يحتاجُ إلى حجّةٍ (علميّةٍ أو واقعيّةٍ أو منطقيّةٍ). وليستِ العبارةُ حقيقةً، والصّورةُ ليستْ ممنوعةً في الحجاجِ.
-- 06#6 [mcq] d3 قالَ تعالى: ﴿وَلَهُ الْجَوَارِ الْمُنشَآتُ فِي الْبَحْرِ كَالْأَعْلَامِ﴾ (سورة الرّحمن). (الجَوارِ المنشآتُ: السّفنُ، والأعلامُ: الجبالُ). أيُّ التّحليلاتِ التّاليةِ يملأُ جدولَ الأركانِ ملأً صحيحًا؟
   a) الأداةُ الكافُ، والمشبَّهُ الجبالُ، والمشبَّهُ به السّفنُ، ووجهُ الشّبَهِ الضّخامةُ والارتفاعُ
  *b) الأداةُ الكافُ، والمشبَّهُ السّفنُ، والمشبَّهُ به الجبالُ، ووجهُ الشّبَهِ الضّخامةُ والارتفاعُ
   c) لا أداةَ فيه، والمشبَّهُ السّفنُ، والمشبَّهُ به الجبالُ، ووجهُ الشّبَهِ الضّخامةُ والارتفاعُ
   d) الأداةُ الكافُ، والمشبَّهُ السّفنُ، والمشبَّهُ به البحرُ، ووجهُ الشّبَهِ الجريانُ والحركةُ
  EXPL: المشبَّهُ هو الموصوفُ «الجَوارِ المنشآتُ» أي السّفنُ، والمشبَّهُ به «الأعلامُ» أي الجبالُ، والكافُ في «كالأعلامِ» هي الأداةُ، ووجهُ الشّبَهِ لم يُذكرْ فنستنتجُه: الضّخامةُ والارتفاعُ. وقلبُ الطّرفَين، أو إنكارُ الأداةِ، أو جعلُ البحرِ مشبَّهًا به، أخطاءٌ في ملءِ الجدولِ.
